    22→    22→function useReveal() {
    23→    23→  const ref = useRef<HTMLDivElement>(null);
    24→    24→  const [v, setV] = useState(false);
    25→    25→  useEffect(() => {
    26→    26→    const el = ref.current;
    27→    27→    if (!el) return;
    28→    28→    const o = new IntersectionObserver(([e]) => { if (e.isIntersecting) { setV(true); o.unobserve(el); } }, { threshold: 0.12 });
    29→    29→    o.observe(el);
    30→    30→    return () => o.disconnect();
    31→    31→  }, []);
    32→    32→  return { ref, visible: v };
    33→    33→}
    34→    34→
    35→    35→function Rv({ children, delay = 0, className = "" }: { children: React.ReactNode; delay?: number; className?: string }) {
    36→    36→  const { ref, visible } = useReveal();
    37→    37→  return (
    38→    38→    <div ref={ref} className={`rv ${visible ? "in" : ""} ${className}`} style={{ transitionDelay: `${delay}ms` }}>
    39→    39→      {children}
    40→    40→    </div>
    41→    41→  );
    42→    42→}

    46→    46→function Countdown() {
    47→    47→  const [t, setT] = useState({ d: 0, h: 0, m: 0, s: 0 });
    48→    48→  const [mounted, setMounted] = useState(false);
    49→    49→  useEffect(() => {
    50→    50→    const tick = () => {
    51→    51→      const diff = EVENT.getTime() - Date.now();
    52→    52→      if (diff <= 0) return;
    53→    53→      setT({ d: Math.floor(diff/864e5), h: Math.floor((diff/36e5)%24), m: Math.floor((diff/6e4)%60), s: Math.floor((diff/1e3)%60) });
    54→    54→    };
    55→    55→    tick();
    56→    56→    setMounted(true);
    57→    57→    const id = setInterval(tick, 1000);
    58→    58→    return () => clearInterval(id);
    59→    59→  }, []);
    60→    60→  return (
    61→    61→    <div className="flex gap-5 sm:gap-8 justify-center">
    62→    62→      {[{v:t.d,l:"Dias"},{v:t.h,l:"Horas"},{v:t.m,l:"Min"},{v:t.s,l:"Seg"}].map(u=>(
    63→    63→        <div key={u.l} className="flex flex-col items-center">
    64→    64→          <span className="text-3xl sm:text-5xl font-extralight tabular-nums tracking-tight" style={{color:"var(--t1)"}} suppressHydrationWarning>
    65→    65→            {mounted ? String(u.v).padStart(2,"0") : "\u2013\u2013"}
    66→    66→          </span>

    68→    68→        </div>

    70→    70→    </div>
    71→    71→  );
    72→    72→}
