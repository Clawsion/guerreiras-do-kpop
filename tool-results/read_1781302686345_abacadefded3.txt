    22→function useReveal() {
    23→  const ref = useRef<HTMLDivElement>(null);
    24→  const [v, setV] = useState(false);
    25→  useEffect(() => {
    26→    const el = ref.current;
    27→    if (!el) return;
    28→    const o = new IntersectionObserver(([e]) => { if (e.isIntersecting) { setV(true); o.unobserve(el); } }, { threshold: 0.12 });
    29→    o.observe(el);
    30→    return () => o.disconnect();
    31→  }, []);
    32→  return { ref, visible: v };
    33→}

    35→function Rv({ children, delay = 0, className = "" }: { children: React.ReactNode; delay?: number; className?: string }) {
    36→  const { ref, visible } = useReveal();

    39→      {children}
    40→    </div>
    41→  );
    42→}

    46→function Countdown() {
    47→  const [t, setT] = useState({ d: 0, h: 0, m: 0, s: 0 });
    48→  const [mounted, setMounted] = useState(false);
    49→  useEffect(() => {
    50→    const tick = () => {
    51→      const diff = EVENT.getTime() - Date.now();
    52→      if (diff <= 0) return;
    53→      setT({ d: Math.floor(diff/864e5), h: Math.floor((diff/36e5)%24), m: Math.floor((diff/6e4)%60), s: Math.floor((diff/1e3)%60) });
    54→    };
    55→    tick();
    56→    setMounted(true);
    57→    const id = setInterval(tick, 1000);
    58→    return () => clearInterval(id);
    59→  }, []);
    60→  return (
    61→    <div className="flex gap-5 sm:gap-8 justify-center">
    62→      {[{v:t.d,l:"Dias"},{v:t.h,l:"Horas"},{v:t.m,l:"Min"},{v:t.s,l:"Seg"}].map(u=>(
    63→        <div key={u.l} className="flex flex-col items-center">
    64→          <span className="text-3xl sm:text-5xl font-extralight tabular-nums tracking-tight" style={{color:"var(--t1)"}} suppressHydrationWarning>
    65→            {mounted ? String(u.v).padStart(2,"0") : "\u2013\u2013"}
    66→          </span>
    67→          <span className="sec-num mt-1.5">{u.l}</span>
    68→        </div>
    69→      ))}
    70→    </div>
    71→  );
    72→}

   106→function LedWallVideo({ active }: { active: boolean }) {
   107→  const vidA = useRef<HTMLVideoElement>(null);
   108→  const vidB = useRef<HTMLVideoElement>(null);
   109→  const sampleCanvas = useRef<HTMLCanvasElement>(null);
   110→  const sectionRef = useRef<HTMLElement | null>(null);
   111→  const startedRef = useRef(false);
   112→  const idxRef = useRef(0);
   113→  const phaseRef = useRef<"A" | "B">("A");
   114→  const rafRef = useRef(0);
   115→  const activeRef = useRef(false);
   116→  const lastSyncRef = useRef(0);
   117→  const [on, setOn] = useState(false);
   118→
   119→  /* Start video after power-on delay */
   120→  useEffect(() => {
   121→    activeRef.current = active;
   122→    if (active && !startedRef.current) {
   123→      const t = setTimeout(() => {
   124→        startedRef.current = true;
   125→        setOn(true);
   126→        const v = vidA.current;
   127→        if (v) {
   128→          v.src = LED_VIDEOS[0];
   129→          v.play().catch(() => {});
   130→        }
   131→      }, 1800);
   132→      return () => clearTimeout(t);
   133→    }
   134→  }, [active]);
   135→
   136→  /* Color sampler — reads dominant color from video and syncs glow */
   137→  useEffect(() => {
   138→    const canvas = sampleCanvas.current;
   139→    if (!canvas) return;
   140→    const ctx = canvas.getContext("2d", { willReadFrequently: true });
   141→    if (!ctx) return;
   142→
   143→    sectionRef.current = canvas.closest(".manifesto-section") as HTMLElement;
   144→
   145→    function sample() {
   146→      if (!startedRef.current || !activeRef.current) {
   147→        rafRef.current = requestAnimationFrame(sample);
   148→        return;
   149→      }
   150→
   151→      const now = performance.now() / 1000;
   152→      const vid = phaseRef.current === "A" ? vidA.current : vidB.current;
   153→
   154→      let synced = false;
   155→      if (vid && vid.readyState >= 2) {
   156→        try {
   157→          canvas.width = 40;
   158→          canvas.height = 22;
   159→          ctx!.drawImage(vid, 0, 0, 40, 22);
   160→          const data = ctx!.getImageData(10, 5, 20, 12).data;
   161→          let r = 0, g = 0, b = 0, n = 0;
   162→          for (let i = 0; i < data.length; i += 4) {
   163→            if (data[i] + data[i+1] + data[i+2] > 40) {
   164→              r += data[i]; g += data[i+1]; b += data[i+2]; n++;
   165→            }
   166→          }
   167→          if (n > 0) {
   168→            r = Math.round(r / n); g = Math.round(g / n); b = Math.round(b / n);
   169→            const el = sectionRef.current;
   170→            if (el && now - lastSyncRef.current > 0.15) {
   171→              lastSyncRef.current = now;
   172→              el.style.setProperty("--led-r", `${r}`);
   173→              el.style.setProperty("--led-g", `${g}`);
   174→              el.style.setProperty("--led-b", `${b}`);
   175→            }
   176→            synced = true;
   177→          }
   178→        } catch { /* tainted canvas fallback below */ }
   179→      }
   180→
   181→      /* Fallback: use pre-set dominant color for current clip */
   182→      if (!synced && now - lastSyncRef.current > 0.3) {
   183→        lastSyncRef.current = now;
   184→        const [r, g, b] = LED_COLORS[idxRef.current % LED_COLORS.length];
   185→        const el = sectionRef.current;
   186→        if (el) {
   187→          el.style.setProperty("--led-r", `${r}`);
   188→          el.style.setProperty("--led-g", `${g}`);
   189→          el.style.setProperty("--led-b", `${b}`);
   190→        }
   191→      }
   192→
   193→      rafRef.current = requestAnimationFrame(sample);
   194→    }
   195→
   196→    rafRef.current = requestAnimationFrame(sample);
   197→    return () => cancelAnimationFrame(rafRef.current);
   198→  }, []);
   199→
   200→  /* Crossfade timer — switches clips on schedule */
   201→  useEffect(() => {
   202→    if (!active) return;
   203→    const interval = setInterval(() => {
   204→      if (!startedRef.current) return;
   205→
   206→      const nextIdx = (idxRef.current + 1) % LED_VIDEOS.length;
   207→      idxRef.current = nextIdx;
   208→      const nextSrc = LED_VIDEOS[nextIdx];
   209→
   210→      if (phaseRef.current === "A") {
   211→        const vb = vidB.current;
   212→        if (vb) { vb.src = nextSrc; vb.play().catch(() => {}); }
   213→        if (vidA.current) vidA.current.style.opacity = "0";
   214→        if (vb) vb.style.opacity = "1";
   215→        phaseRef.current = "B";
   216→      } else {
   217→        const va = vidA.current;
   218→        if (va) { va.src = nextSrc; va.play().catch(() => {}); }
   219→        if (vidB.current) vidB.current.style.opacity = "0";
   220→        if (va) va.style.opacity = "1";
   221→        phaseRef.current = "A";
   222→      }
   223→    }, CLIP_DURATION * 1000);
   224→
   225→    return () => clearInterval(interval);
   226→  }, [active]);
   227→
   228→  return (
   229→    <div className="led-video-container">
   230→      <video
   231→        ref={vidA}
   232→        className="led-video-layer"
   233→        muted
   234→        loop
   235→        playsInline
   236→        preload="auto"
   237→        crossOrigin="anonymous"
   238→        style={{ opacity: on ? 1 : 0 }}
   239→      />
   240→      <video
   241→        ref={vidB}
   242→        className="led-video-layer"
   243→        muted
   244→        loop
   245→        playsInline
   246→        preload="auto"
   247→        crossOrigin="anonymous"
   248→        style={{ opacity: 0 }}
   249→      />
   250→      <canvas ref={sampleCanvas} style={{ display: "none" }} />
   251→    </div>
   252→  );
   253→}
