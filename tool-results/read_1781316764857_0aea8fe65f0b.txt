    22→function useReveal() {
    23→  const ref = useRef<HTMLDivElement>(null);
    24→  const [v, setV] = useState(false);
    25→  useEffect(() => {
    26→    const el = ref.current;
    27→    if (!el) return;
    28→    const o = new IntersectionObserver(([e]) => { if (e.isIntersecting) { setV(true); o.unobserve(el); } }, { threshold: 0.12 });
    29→    o.observe(el);
    30→    return () => o.disconnect();
    31→  }, []);
    32→  return { ref, visible: v };
    33→}

    35→function Rv({ children, delay = 0, className = "" }: { children: React.ReactNode; delay?: number; className?: string }) {
    36→  const { ref, visible } = useReveal();

    38→    <div ref={ref} className={`rv ${visible ? "in" : ""} ${className}`} style={{ transitionDelay: `${delay}ms` }}>
    39→      {children}
    40→    </div>
    41→  );
    42→}

    45→
    46→function Countdown() {
    47→  const [t, setT] = useState({ d: 0, h: 0, m: 0, s: 0 });
    48→  const [mounted, setMounted] = useState(false);
    49→  useEffect(() => {
    50→    const tick = () => {
    51→      const diff = EVENT.getTime() - Date.now();
    52→      if (diff <= 0) return;
    53→      setT({ d: Math.floor(diff/864e5), h: Math.floor((diff/36e5)%24), m: Math.floor((diff/6e4)%60), s: Math.floor((diff/1e3)%60) });
    54→    };
    55→    tick();
    56→    setMounted(true);
    57→    const id = setInterval(tick, 1000);
    58→    return () => clearInterval(id);
    59→  }, []);
    60→  return (
    61→    <div className="flex gap-5 sm:gap-8 justify-center">
    62→      {[{v:t.d,l:"Dias"},{v:t.h,l:"Horas"},{v:t.m,l:"Min"},{v:t.s,l:"Seg"}].map(u=>(
    63→        <div key={u.l} className="flex flex-col items-center">
    64→          <span className="text-3xl sm:text-5xl font-extralight tabular-nums tracking-tight" style={{color:"var(--t1)"}} suppressHydrationWarning>
    65→            {mounted ? String(u.v).padStart(2,"0") : "\u2013\u2013"}
    66→          </span>
    67→          <span className="sec-num mt-1.5">{u.l}</span>
    68→        </div>
    69→      ))}
    70→    </div>
    71→  );
    72→}

   104→function LedWallVideo({ active }: { active: boolean }) {
   105→  const vidRef = useRef<HTMLVideoElement>(null);
   106→  const sampleCanvas = useRef<HTMLCanvasElement>(null);
   107→  const sectionRef = useRef<HTMLElement | null>(null);
   108→  const startedRef = useRef(false);
   109→  const rafRef = useRef(0);
   110→  const activeRef = useRef(false);
   111→  const lastSyncRef = useRef(0);
   112→  const [on, setOn] = useState(false);
   113→
   114→  /* Start video after power-on delay */
   115→  useEffect(() => {
   116→    activeRef.current = active;
   117→    if (active && !startedRef.current) {
   118→      const t = setTimeout(() => {
   119→        startedRef.current = true;
   120→        setOn(true);
   121→        const v = vidRef.current;
   122→        if (v) {
   123→          v.play().catch(() => {});
   124→        }
   125→      }, 1800);
   126→      return () => clearTimeout(t);
   127→    }
   128→  }, [active]);
   129→
   130→  /* Color sampler — reads dominant color from video and syncs glow */
   131→  useEffect(() => {
   132→    const canvas = sampleCanvas.current;
   133→    if (!canvas) return;
   134→    const ctx = canvas.getContext("2d", { willReadFrequently: true });
   135→    if (!ctx) return;
   136→
   137→    sectionRef.current = canvas.closest(".manifesto-section") as HTMLElement;
   138→
   139→    function sample() {
   140→      if (!startedRef.current || !activeRef.current) {
   141→        rafRef.current = requestAnimationFrame(sample);
   142→        return;
   143→      }
   144→
   145→      const now = performance.now() / 1000;
   146→      const vid = vidRef.current;
   147→
   148→      let synced = false;
   149→      if (vid && vid.readyState >= 2) {
   150→        try {
   151→          canvas.width = 40;
   152→          canvas.height = 22;
   153→          ctx!.drawImage(vid, 0, 0, 40, 22);
   154→          const data = ctx!.getImageData(10, 5, 20, 12).data;
   155→          let r = 0, g = 0, b = 0, n = 0;
   156→          for (let i = 0; i < data.length; i += 4) {
   157→            if (data[i] + data[i+1] + data[i+2] > 40) {
   158→              r += data[i]; g += data[i+1]; b += data[i+2]; n++;
   159→            }
   160→          }
   161→          if (n > 0) {
   162→            r = Math.round(r / n); g = Math.round(g / n); b = Math.round(b / n);
   163→            const el = sectionRef.current;
   164→            if (el && now - lastSyncRef.current > 0.15) {
   165→              lastSyncRef.current = now;
   166→              el.style.setProperty("--led-r", `${r}`);
   167→              el.style.setProperty("--led-g", `${g}`);
   168→              el.style.setProperty("--led-b", `${b}`);
   169→            }
   170→            synced = true;
   171→          }
   172→        } catch { /* tainted canvas fallback below */ }
   173→      }
   174→
   175→      /* Fallback: cycle through tunnel color palette */
   176→      if (!synced && now - lastSyncRef.current > 0.3) {
   177→        lastSyncRef.current = now;
   178→        const idx = Math.floor(now / 3) % TUNNEL_COLORS.length;
   179→        const [r, g, b] = TUNNEL_COLORS[idx];
   180→        const el = sectionRef.current;
   181→        if (el) {
   182→          el.style.setProperty("--led-r", `${r}`);
   183→          el.style.setProperty("--led-g", `${g}`);
   184→          el.style.setProperty("--led-b", `${b}`);
   185→        }
   186→      }
   187→
   188→      rafRef.current = requestAnimationFrame(sample);
   189→    }
   190→
   191→    rafRef.current = requestAnimationFrame(sample);
   192→    return () => cancelAnimationFrame(rafRef.current);
   193→  }, []);
   194→
   195→  return (
   196→    <div className="led-video-container">
   197→      <video
   198→        ref={vidRef}
   199→        className="led-video-layer"
   200→        src={LED_TUNNEL_SRC}
   201→        muted
   202→        loop
   203→        playsInline
   204→        preload="auto"
   205→        style={{ opacity: on ? 1 : 0 }}
   206→      />
   207→      <canvas ref={sampleCanvas} style={{ display: "none" }} />
   208→    </div>
   209→  );
   210→}
