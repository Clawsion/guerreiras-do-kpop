    21→
    22→function useReveal() {
    23→  const ref = useRef<HTMLDivElement>(null);
    24→  const [v, setV] = useState(false);
    25→  useEffect(() => {
    26→    const el = ref.current;
    27→    if (!el) return;
    28→    const o = new IntersectionObserver(([e]) => { if (e.isIntersecting) { setV(true); o.unobserve(el); } }, { threshold: 0.12 });
    29→    o.observe(el);
    30→    return () => o.disconnect();
    31→  }, []);
    32→  return { ref, visible: v };
    33→}
    34→
    35→function Rv({ children, delay = 0, className = "" }: { children: React.ReactNode; delay?: number; className?: string }) {
    36→  const { ref, visible } = useReveal();
    37→  return (
    38→    <div ref={ref} className={`rv ${visible ? "in" : ""} ${className}`} style={{ transitionDelay: `${delay}ms` }}>
    39→      {children}
    40→    </div>
    41→  );
    42→}

    45→
    46→function Countdown() {
    47→  const [t, setT] = useState({ d: 0, h: 0, m: 0, s: 0 });
    48→  const [mounted, setMounted] = useState(false);
    49→  useEffect(() => {
    50→    const tick = () => {
    51→      const diff = EVENT.getTime() - Date.now();
    52→      if (diff <= 0) return;
    53→      setT({ d: Math.floor(diff/864e5), h: Math.floor((diff/36e5)%24), m: Math.floor((diff/6e4)%60), s: Math.floor((diff/1e3)%60) });
    54→    };
    55→    tick();
    56→    setMounted(true);
    57→    const id = setInterval(tick, 1000);
    58→    return () => clearInterval(id);
    59→  }, []);
    60→  return (
    61→    <div className="flex gap-5 sm:gap-8 justify-center">
    62→      {[{v:t.d,l:"Dias"},{v:t.h,l:"Horas"},{v:t.m,l:"Min"},{v:t.s,l:"Seg"}].map(u=>(
    63→        <div key={u.l} className="flex flex-col items-center">
    64→          <span className="text-3xl sm:text-5xl font-extralight tabular-nums tracking-tight" style={{color:"var(--t1)"}} suppressHydrationWarning>
    65→            {mounted ? String(u.v).padStart(2,"0") : "\u2013\u2013"}
    66→          </span>
    67→          <span className="sec-num mt-1.5">{u.l}</span>
    68→        </div>
    69→      ))}
    70→    </div>
    71→  );
    72→}

    88→/* ═══ LED WALL CANVAS — neon tunnel animation with sync reflections ═══ */
    89→
    90→function LedWallCanvas({ active }: { active: boolean }) {
    91→  const canvasRef = useRef<HTMLCanvasElement>(null);
    92→  const rafRef = useRef(0);
    93→  const startedRef = useRef(false);
    94→  const timeRef = useRef(0);
    95→  const sectionRef = useRef<HTMLElement | null>(null);
    96→  const activeRef = useRef(false);
    97→
    98→  useEffect(() => {
    99→    activeRef.current = active;
   100→    if (active && !startedRef.current) {
   101→      const t = setTimeout(() => { startedRef.current = true; }, 1800);
   102→      return () => clearTimeout(t);
   103→    }
   104→  }, [active]);
   105→
   106→  useEffect(() => {
   107→    const canvas = canvasRef.current;
   108→    if (!canvas) return;
   109→    const ctx = canvas.getContext("2d");
   110→    if (!ctx) return;
   111→
   112→    sectionRef.current = canvas.closest(".manifesto-section") as HTMLElement;
   113→
   114→    const palettes = [
   115→      ["#C850FF","#FF2D78","#818CF8","#B1D5F0","#A855F7"],
   116→      ["#8B5CF6","#6366F1","#FF6BA0","#C850FF","#FFF5A4"],
   117→      ["#FF2D78","#F472B6","#C850FF","#6366F1","#B1D5F0"],
   118→      ["#6366F1","#818CF8","#B1D5F0","#C850FF","#FF2D78"],
   119→      ["#A855F7","#FFF5A4","#FF2D78","#8B5CF6","#6366F1"],
   120→    ];
   121→
   122→    const h2r = (h: string) => ({
   123→      r: parseInt(h.slice(1,3),16), g: parseInt(h.slice(3,5),16), b: parseInt(h.slice(5,7),16)
   124→    });
   125→    const mix = (a: number, b: number, t: number) => a + (b - a) * t;
   126→    const colAt = (t: number, off: number) => {
   127→      const pi = Math.floor(t * 0.04) % palettes.length;
   128→      const ni = (pi + 1) % palettes.length;
   129→      const ci = Math.floor(t * 0.12 + off) % 5;
   130→      const a = h2r(palettes[pi][ci]), b = h2r(palettes[ni][ci]);
   131→      const bt = (t * 0.04) % 1;
   132→      return { r: Math.round(mix(a.r,b.r,bt)), g: Math.round(mix(a.g,b.g,bt)), b: Math.round(mix(a.b,b.b,bt)) };
   133→    };
   134→
   135→    let lastSync = 0;
   136→
   137→    function resize() {
   138→      const r = canvas!.getBoundingClientRect();
   139→      canvas!.width = Math.floor(r.width);
   140→      canvas!.height = Math.floor(r.height);
   141→    }
   142→    resize();
   143→    window.addEventListener("resize", resize);
   144→
   145→    function draw() {
   146→      const w = canvas!.width, h = canvas!.height;
   147→      const cx = w / 2, cy = h / 2;
   148→      const maxR = Math.sqrt(cx*cx + cy*cy);
   149→
   150→      if (!startedRef.current) {
   151→        ctx!.fillStyle = "#050208";
   152→        ctx!.fillRect(0,0,w,h);
   153→        if (activeRef.current && Math.random() < 0.03) {
   154→          ctx!.fillStyle = `rgba(200,80,255,${0.04 + Math.random()*0.1})`;
   155→          ctx!.fillRect(0,0,w,h);
   156→        }
   157→        rafRef.current = requestAnimationFrame(draw);
   158→        return;
   159→      }
   160→
   161→      const t = timeRef.current;
   162→      ctx!.fillStyle = "#050208";
   163→      ctx!.fillRect(0,0,w,h);
   164→
   165→      /* ── Primary tunnel: rings expanding outward ── */
   166→      const rings = 35, speed = t * 100, sp = maxR / rings;
   167→      for (let i = 0; i < rings; i++) {
   168→        const r = i * sp + (speed % sp);
   169→        if (r > maxR || r < 3) continue;
   170→        const p = r / maxR, a = 1 - p;
   171→        if (a < 0.01) continue;
   172→        const c = colAt(t, i * 0.6);
   173→
   174→        /* glow */
   175→        ctx!.beginPath(); ctx!.arc(cx,cy,r,0,Math.PI*2);
   176→        ctx!.strokeStyle = `rgba(${c.r},${c.g},${c.b},${a*0.16})`;
   177→        ctx!.lineWidth = 14 + a*20; ctx!.stroke();
   178→
   179→        /* ring */
   180→        ctx!.beginPath(); ctx!.arc(cx,cy,r,0,Math.PI*2);
   181→        ctx!.strokeStyle = `rgba(${c.r},${c.g},${c.b},${a*0.78})`;
   182→        ctx!.lineWidth = 1.5 + a*3.5; ctx!.stroke();
   183→      }
   184→
   185→      /* ── Secondary tunnel: rings contracting inward ── */
   186→      const rings2 = 18, speed2 = t * 60;
   187→      for (let i = 0; i < rings2; i++) {
   188→        const baseR = maxR - i * (maxR / rings2) - (speed2 % (maxR / rings2));
   189→        if (baseR < 3 || baseR > maxR) continue;
   190→        const p = baseR / maxR, a = (1 - p) * 0.35;
   191→        if (a < 0.01) continue;
   192→        const c = colAt(t + 5, i * 0.8 + 2);
   193→
   194→        ctx!.beginPath(); ctx!.arc(cx,cy,baseR,0,Math.PI*2);
   195→        ctx!.strokeStyle = `rgba(${c.r},${c.g},${c.b},${a*0.5})`;
   196→        ctx!.lineWidth = 1 + a*3; ctx!.stroke();
   197→      }
   198→
   199→      /* ── Rotating neon beams ── */
   200→      const beams = 6;
   201→      for (let i = 0; i < beams; i++) {
   202→        const ang = (i / beams) * Math.PI * 2 + t * 0.22;
   203→        const c = colAt(t, i * 1.3);
   204→        const oR = maxR * 0.75;
   205→        const grad = ctx!.createLinearGradient(
   206→          cx + Math.cos(ang)*15, cy + Math.sin(ang)*15,
   207→          cx + Math.cos(ang)*oR, cy + Math.sin(ang)*oR
   208→        );
   209→        grad.addColorStop(0, `rgba(${c.r},${c.g},${c.b},0.28)`);
   210→        grad.addColorStop(0.35, `rgba(${c.r},${c.g},${c.b},0.05)`);
   211→        grad.addColorStop(1, `rgba(${c.r},${c.g},${c.b},0)`);
   212→
   213→        ctx!.beginPath();
   214→        ctx!.moveTo(cx+Math.cos(ang-0.025)*15, cy+Math.sin(ang-0.025)*15);
   215→        ctx!.lineTo(cx+Math.cos(ang-0.06)*oR, cy+Math.sin(ang-0.06)*oR);
   216→        ctx!.lineTo(cx+Math.cos(ang+0.06)*oR, cy+Math.sin(ang+0.06)*oR);
   217→        ctx!.lineTo(cx+Math.cos(ang+0.025)*15, cy+Math.sin(ang+0.025)*15);
   218→        ctx!.closePath();
   219→        ctx!.fillStyle = grad; ctx!.fill();
   220→      }
   221→
   222→      /* ── Horizontal sweep line ── */
   223→      const sweepY = ((t * 40) % (h + 60)) - 30;
   224→      const dc = colAt(t, 0);
   225→      const sweepGrad = ctx!.createLinearGradient(0, sweepY - 25, 0, sweepY + 25);
   226→      sweepGrad.addColorStop(0, `rgba(${dc.r},${dc.g},${dc.b},0)`);
   227→      sweepGrad.addColorStop(0.5, `rgba(${dc.r},${dc.g},${dc.b},0.08)`);
   228→      sweepGrad.addColorStop(1, `rgba(${dc.r},${dc.g},${dc.b},0)`);
   229→      ctx!.fillStyle = sweepGrad;
   230→      ctx!.fillRect(0, sweepY - 25, w, 50);
   231→
   232→      /* ── Center bloom ── */
   233→      const cg = ctx!.createRadialGradient(cx,cy,0,cx,cy,maxR*0.28);
   234→      cg.addColorStop(0, `rgba(${dc.r},${dc.g},${dc.b},0.14)`);
   235→      cg.addColorStop(0.5, `rgba(${dc.r},${dc.g},${dc.b},0.03)`);
   236→      cg.addColorStop(1, "transparent");
   237→      ctx!.fillStyle = cg; ctx!.fillRect(0,0,w,h);
   238→
   239→      /* ── Pulse flash ── */
   240→      const pt = t % 6;
   241→      if (pt < 0.25) {
   242→        const pa = Math.sin(pt/0.25*Math.PI)*0.12;
   243→        ctx!.fillStyle = `rgba(${dc.r},${dc.g},${dc.b},${pa})`;
   244→        ctx!.fillRect(0,0,w,h);
   245→      }
   246→
   247→      /* ── Second offset pulse ── */
   248→      const pt2 = (t+3) % 6;
   249→      if (pt2 < 0.18) {
   250→        const c2 = colAt(t+3, 2);
   251→        const pa2 = Math.sin(pt2/0.18*Math.PI)*0.08;
   252→        ctx!.fillStyle = `rgba(${c2.r},${c2.g},${c2.b},${pa2})`;
   253→        ctx!.fillRect(0,0,w,h);
   254→      }
   255→
   256→      /* ── Sync reflection color ── */
   257→      if (t - lastSync > 0.13) {
   258→        lastSync = t;
   259→        const el = sectionRef.current;
   260→        if (el) {
   261→          el.style.setProperty("--led-r", `${dc.r}`);
   262→          el.style.setProperty("--led-g", `${dc.g}`);
   263→          el.style.setProperty("--led-b", `${dc.b}`);
   264→        }
   265→      }
   266→
   267→      timeRef.current += 0.016;
   268→      rafRef.current = requestAnimationFrame(draw);
   269→    }
   270→
   271→    rafRef.current = requestAnimationFrame(draw);
   272→    return () => { cancelAnimationFrame(rafRef.current); window.removeEventListener("resize", resize); };
   273→  }, []);
   274→
   275→  return <canvas ref={canvasRef} className="led-canvas"/>;
   276→}
