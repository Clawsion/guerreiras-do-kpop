    22→    22→function useReveal() {
    23→    23→  const ref = useRef<HTMLDivElement>(null);
    24→    24→  const [v, setV] = useState(false);
    25→    25→  useEffect(() => {
    26→    26→    const el = ref.current;
    27→    27→    if (!el) return;
    28→    28→    const o = new IntersectionObserver(([e]) => { if (e.isIntersecting) { setV(true); o.unobserve(el); } }, { threshold: 0.12 });
    29→    29→    o.observe(el);
    30→    30→    return () => o.disconnect();
    31→    31→  }, []);
    32→    32→  return { ref, visible: v };
    33→    33→}
    34→    34→
    35→    35→function Rv({ children, delay = 0, className = "" }: { children: React.ReactNode; delay?: number; className?: string }) {
    36→    36→  const { ref, visible } = useReveal();
    37→    37→  return (
    38→    38→    <div ref={ref} className={`rv ${visible ? "in" : ""} ${className}`} style={{ transitionDelay: `${delay}ms` }}>
    39→    39→      {children}
    40→    40→    </div>
    41→    41→  );
    42→    42→}

    48→    48→  useEffect(() => {
    49→    49→    const tick = () => {
    50→    50→      const diff = EVENT.getTime() - Date.now();
    51→    51→      if (diff <= 0) return;
    52→    52→      setT({ d: Math.floor(diff/864e5), h: Math.floor((diff/36e5)%24), m: Math.floor((diff/6e4)%60), s: Math.floor((diff/1e3)%60) });
    53→    53→    };
    54→    54→    tick();
    55→    55→    const id = setInterval(tick, 1000);
    56→    56→    return () => clearInterval(id);
    57→    57→  }, []);
    58→    58→  return (
    59→    59→    <div className="flex gap-5 sm:gap-8 justify-center">
