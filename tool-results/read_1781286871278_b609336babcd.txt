   181→function CountdownTimer() {
   182→  const [timeLeft, setTimeLeft] = useState({
   183→    days: 0,
   184→    hours: 0,
   185→    minutes: 0,
   186→    seconds: 0,
   187→  });
   188→
   189→  useEffect(() => {
   190→    const tick = () => {
   191→      const now = new Date();
   192→      const diff = EVENT_DATE.getTime() - now.getTime();
   193→      if (diff <= 0) return;
   194→      setTimeLeft({
   195→        days: Math.floor(diff / (1000 * 60 * 60 * 24)),
   196→        hours: Math.floor((diff / (1000 * 60 * 60)) % 24),
   197→        minutes: Math.floor((diff / (1000 * 60)) % 60),
   198→        seconds: Math.floor((diff / 1000) % 60),
   199→      });
   200→    };
   201→    tick();
   202→    const interval = setInterval(tick, 1000);
   203→    return () => clearInterval(interval);
   204→  }, []);
   205→
   206→  const units = [
   207→    { label: "Dias", value: timeLeft.days },
   208→    { label: "Horas", value: timeLeft.hours },
   209→    { label: "Min", value: timeLeft.minutes },
   210→    { label: "Seg", value: timeLeft.seconds },
   211→  ];
   212→
   213→  return (
   214→    <div className="flex gap-3 sm:gap-4 justify-center">
   215→      {units.map((u) => (
   216→        <div key={u.label} className="flex flex-col items-center">
   217→          <div className="glass-card rounded-xl px-3 py-2 sm:px-5 sm:py-3 min-w-[60px] sm:min-w-[80px] text-center">
   218→            <span className="text-2xl sm:text-4xl font-bold text-yellow-400 neon-text-gold font-mono">
   219→              {String(u.value).padStart(2, "0")}
   220→            </span>
   221→          </div>
   222→          <span className="text-xs sm:text-sm text-muted-foreground mt-1.5 uppercase tracking-wider">
   223→            {u.label}
   224→          </span>
   225→        </div>
   226→      ))}
   227→    </div>
   228→  );
   229→}

   231→function AnimatedSection({
   232→  children,
   233→  className = "",
   234→  delay = 0,
   235→}: {
   236→  children: React.ReactNode;
   237→  className?: string;
   238→  delay?: number;
   239→}) {
   240→  const ref = useRef(null);
   241→  const isInView = useInView(ref, { once: true, margin: "-80px" });
   242→
   243→  return (
   244→    <motion.div
   245→      ref={ref}
   246→      initial={{ opacity: 0, y: 50 }}
   247→      animate={isInView ? { opacity: 1, y: 0 } : { opacity: 0, y: 50 }}
   248→      transition={{ duration: 0.7, delay, ease: "easeOut" }}
   249→      className={className}
   250→    >
   251→      {children}
   252→    </motion.div>
   253→  );
   254→}
   255→
   256→function SparkleEffect() {
   257→  const sparkles = Array.from({ length: 20 }, (_, i) => ({
   258→    id: i,
   259→    left: `${Math.random() * 100}%`,
   260→    top: `${Math.random() * 100}%`,
   261→    size: Math.random() * 4 + 2,
   262→    delay: Math.random() * 4,
   263→    duration: Math.random() * 2 + 2,
   264→  }));
   265→
   266→  return (
   267→    <div className="absolute inset-0 overflow-hidden pointer-events-none">
   268→      {sparkles.map((s) => (
   269→        <div
   270→          key={s.id}
   271→          className="absolute rounded-full bg-yellow-400/60"
   272→          style={{
   273→            left: s.left,
   274→            top: s.top,
   275→            width: s.size,
   276→            height: s.size,
   277→            animation: `sparkle ${s.duration}s ease-in-out ${s.delay}s infinite`,
   278→          }}
   279→        />
   280→      ))}
   281→    </div>
   282→  );
   283→}
